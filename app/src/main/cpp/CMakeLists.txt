cmake_minimum_required(VERSION 3.22.1)
project(luxio C)
file(GLOB LUA_SOURCES lua/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(luxio SHARED AnimationRuntime.c AndroidRuntime.c ${LUA_SOURCES})
target_include_directories(luxio PRIVATE lua)
target_compile_definitions(luxio PRIVATE LUA_USE_POSIX)
target_link_libraries(luxio m)
target_link_options(luxio PRIVATE "-Wl,-z,max-page-size=16384")
